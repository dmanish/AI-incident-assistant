def get_openai_client():
    """Lazy-loads OpenAI client, handles version mismatches"""
    global _openai_client
    if _openai_client is not None:
        return _openai_client
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        return None
    _openai_client = OpenAI(api_key=api_key)
    return _openai_client

def query_failed_logins(date_iso: str, username: Optional[str] = None, limit: int = 200):
    """File-backed DuckDB, short-lived connection."""
    log_dir = ROOT / "data" / "logs" / "auth"
    files = list(log_dir.glob("*.csv"))
    if not files:
        return pd.DataFrame([])

    con = duckdb.connect(DUCK_DB_PATH, read_only=False)
    try:
        sql = f"""
            SELECT timestamp, user, action, result, ip
            FROM read_csv_auto('{(log_dir / "*.csv").as_posix()}', union_by_name=true)
            WHERE cast(timestamp as varchar) LIKE ?
              AND lower(result) = 'failed'
              {user_clause}
            ORDER BY timestamp DESC
            LIMIT ?
        """
        df = con.execute(sql, params).fetchdf()
        return df
    finally:
        con.close()

def get_embedder():
    provider = os.getenv("EMBEDDINGS_PROVIDER", "local") or "local"

    if provider == "openai":
        # OpenAI text-embedding-3-small (requires API key)
        return OpenAIv1EmbeddingFunction(api_key, model_name="text-embedding-3-small")
    else:
        # Default: Local CPU embeddings
        return SentenceTransformerEmbeddingFunction(model_name="all-MiniLM-L6-v2")

Document Chunking Strategy

Location: scripts/bootstrap.py:108-113

Function: chunks(text, size=800, overlap=120)

Chunking Parameters:
  • Chunk Size: 800 word tokens
  • Overlap: 120 tokens (15% overlap)
  • Rationale: Maintains context continuity across chunk boundaries
def chunks(text: str, size: int = 800, overlap: int = 120):
    toks = text.split()
    i = 0
    while i < len(toks):
        yield " ".join(toks[i:i+size])
        i += max(1, size - overlap)

    """Tolerant retrieval with RBAC filtering"""
    if collection is None:

    try:
        res = collection.query(

            include=["documents", "metadatas", "distances"],
        )
    except Exception as e:
        audit({"action": "retrieval_error", "error": str(e)})
        return []

    out = []
    for i in range(min(len(docs), len(metas))):

  • "override the system"
